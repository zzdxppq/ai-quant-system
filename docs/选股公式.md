## 通达信选股公式
ZT0:=IF(NAMELIKE('ST') OR NAMELIKE('*ST'),C=ZTPRICE(REF(C,1),0.05),
IF(CODELIKE('688') OR (CODELIKE('300') AND DATE>=1200824),C=ZTPRICE(REF(C,1),0.2),
C=ZTPRICE(REF(C,1),0.1)));

竞价涨幅:=(O/REF(C,1)-1)*100;
 竞换手:=DYNAINFO(15)/O/FINANCE(46)*100;
ZFF1:=O-REF(C,1);
ZFF2:=ZFF1/REF(C,1);
ZFF:=ZFF2*100;
JJL:=DYNAINFO(15)/DYNAINFO(4)/100;
ZTX01:=DATE-1000000;
ZTX02:=ZTX01<200823;
ZTX03:=IF(ZTX02,1.1,1.2);
ZNNX:=IF(CODELIKE('30')<>0,ZTX03,IF(CODELIKE('68')<>0,1.2,1.1));
ZTX1:=(H+0.01);
ZTX2:=REF(C,1);
ZTX3:=H=C;
ZTX4:=ZTX1/ZTX2>=ZNNX AND ZTX3;
PD1:=JJL>1000;
VAR2:=JJL/CAPITAL*100;
PD2:=VAR2>0.5;
PD3:=ZFF>4 AND ZFF<7.5;
VAR3:=CAPITAL*C*100/100000000;
PD4:=VAR3<100;
VAR4:=REF(ZTX4,1);
PD5:=VAR4;
VAR5:=NAMEINCLUDE('ST')=0 AND CODELIKE('68')=0 AND CODELIKE('30')=0 AND CODELIKE('8')=0 AND CODELIKE('4')=0;
LIANBAN:=REF(ZT0,1) AND REF(ZT0,2);
XG:LIANBAN AND PD1 AND PD2 AND PD3 AND PD4 AND PD5 AND VAR5 AND  DYNAINFO(17)>1 AND 竞换手>0.5 AND 竞价涨幅<7.5;
